# 第四部分：核心类 FaultKnowledgeBase

# ==========================================

class FaultKnowledgeBase:

    """

    故障注入知识库：一个完整的 RAG 系统

   

    建库流程：

    1. 读取 Markdown 故障手册

    2. 按标题切分成小块

    3. 每块用 BGE-M3 生成向量

    4. 把向量+原文+元数据存进 Milvus 数据库

   

    查询流程：

    1. 用户提问

    2. BM25 关键词检索（路1）+ Milvus 向量检索（路2）

    3. 合并去重

    4. CrossEncoder 重排精选

    5. 返回最相关的文档给大模型

    """

   

    def __init__(self,

                 md_path: str = DOC_DIR,              # Markdown 故障手册的文件路径

                 db_uri: str = "http://127.0.0.1:19530",  # Milvus 数据库的连接地址（默认本机 19530 端口）

                 collection_name: str = "fault_inject",  # 集合名（类比 MySQL 的表名）

                 force_rebuild: bool = False):      # 是否强制重建数据库（md 文件更新后需要设为 True）

       

        # ----- 1. 确定文件路径 -----

        self.md_path = os.path.join(md_path, "fault_inject.md")

        self.db_uri = db_uri                    # 保存数据库连接地址

        self.collection_name = collection_name  # 保存集合名

        self._ensure_milvus_running() # 在加载模型和连接数据库之前，先唤醒沉睡的容器



        # ----- 2. 加载 Embedding 模型（向量化模型）-----

        print("[RAG] 正在加载 BGE-M3 向量化模型...")

        self.embeddings = HuggingFaceEmbeddings(

            model_name="BAAI/bge-m3",            # 模型名称：北京智源的 BGE-M3，支持中英文，1024维输出

            model_kwargs={'device': 'cpu'},      # 用 CPU 推理（没有 GPU 或 GPU 不够时的选择）

            encode_kwargs={'normalize_embeddings': True}  # 输出向量做 L2 归一化（长度变为1）

        )

        self.embedding_dim = 1024  # BGE-M3 模型输出的向量维度是 1024（这个值由模型架构决定，不能改）



        # ----- 3. 加载 Reranker 模型（重排模型）-----

        print("[RAG] 正在加载 BGE-Reranker-v2-M3 重排模型...")

        # 【核心概念：CrossEncoder（交叉编码器）】

        # 与 Embedding 模型的本质区别：

        #   Embedding：query 和 doc 分别过模型 → 各自得到一个向量 → 算余弦距离（双塔模型）

        #   CrossEncoder：把 [query, doc] 拼成一个整体喂给模型 → 直接输出一个相关性分数（单塔模型）

        self.cross_encoder = CrossEncoder(

            model_name_or_path="BAAI/bge-reranker-v2-m3",  # 北京智源的重排模型，专门给文档相关性打分

            device="cpu",       # 用 CPU 推理

            max_length=1024     # 输入文本最大长度（超过会被截断）

            # 中文大约 1 字 ≈ 1-2 token

        )

       

        # ----- 4. 加载并切分文档 -----

        self.docs = []  # 初始化一个空列表，用来存切分后的所有文档块（Document 对象）

        self._load_and_split_docs()

       

        # ----- 5. 连接 Milvus 数据库并初始化 -----

        print(f"[RAG] 正在连接企业级 Milvus 数据库 ({self.db_uri})...")

        self.client = MilvusClient(uri=self.db_uri)

        self._smart_init_milvus(force_rebuild) # 调用智能初始化方法：判断是建新表还是复用旧表



    # ==========================================

    # 【修改】：Docker 状态检查与自启守护 (Sudo 提权 + 修复 MinIO 名称)

    # ==========================================

    def _ensure_milvus_running(self):

        print("[RAG] 正在检查 Milvus 底层容器状态...")

        try:

            # 【修复 1】: 执行 inspect

            result = subprocess.run(

                ["docker", "inspect", "-f", "{{.State.Running}}", "milvus-standalone"],

                capture_output=True, text=True

            )

           

            # 如果返回值不是 true，说明容器退出或未启动

            if "true" not in result.stdout.lower():

                print("[RAG] 发现 Milvus 容器组未运行，正在使用 sudo 自动唤醒...")

               

                # 【修复 2】: 提前加入 docker 用户组，并将 minio 替换为你机器上确切的容器 ID: 7136498c870f

                subprocess.run(

                    ["docker", "start", "milvus-etcd", "milvus-minio", "milvus-standalone"],

                    check=True

                )

                print("[RAG] 容器组已拉起，正在等待 19530 端口服务就绪...")

               

                # 轮询探测端口 (polling)

                port_ready = False

                for i in range(20): # 最多等待 20 秒

                    try:

                        # 尝试建立 Socket 握手

                        with socket.create_connection(("127.0.0.1", 19530), timeout=1): # 立刻挂断 (Knock and Run)

                            port_ready = True

                            print("[RAG] 端口 19530 通信握手成功，Milvus 服务已完全就绪！")

                            break

                    except OSError:

                        time.sleep(1) # 失败了就等 1s

                        print(f"  ... 内部服务初始化中 ({i+1}/20)")

                       

                if not port_ready:

                    print("[RAG] ⚠️ 警告：等待 Milvus 端口就绪超时，接下来的连接可能会失败。")

            else:

                print("[RAG] Milvus 容器组运行正常，无需唤醒。")

               

        except FileNotFoundError:

            print("[RAG] ❌ 错误：未找到 sudo 或 docker 命令，请确认宿主机环境。")

        except subprocess.CalledProcessError as e:

            print(f"[RAG] ❌ 启动 Docker 容器失败，错误码 {e.returncode}。")

            print("如果终端卡住，可能是 sudo 在等待你输入密码！")



    # ==========================================

    # 文档切分方法

    # ==========================================

    def _load_and_split_docs(self):

        """

        【文档切分 + 上下文增强】



        为什么要切分？

        - 大模型的上下文窗口有限（比如 4096 token），不能把整本手册一次性塞进去

        - 向量化的粒度也要适中：太长 → 向量表征不精确（一个向量描述太多内容会"稀释"语义）

                                太短 → 信息不完整（比如只有半句话，查出来也没用）

        - 按标题切分是 Markdown 文档的最佳实践，每个 # 标题下的内容自然形成一个完整的"知识单元"

       

        为什么要增强上下文？

        - 切分后，标题信息只保留在 metadata 里，正文里可能没有提到故障名称

        - 比如 # CPU过载 下面写着"设置参数为80%"，切出来的正文就是"设置参数为80%"

        - 如果用户搜"CPU"，这段正文里没有"CPU"这个词，BM25 就搜不到！

        - 所以我们把标题强行拼到正文开头，让每段文本都"自带身份证"

        """

        # 读取整个 Markdown 文件的全部内容到内存中（一个大字符串）

        with open(self.md_path, "r", encoding="utf-8") as f:

            md_content = f.read()



        headers_to_split_on = [("#", "fault_name")] # 按照 '#' 切分, 标题作为元数据的 'fault_name' 字段

       

        # 创建 Markdown 切分器实例

        markdown_splitter = MarkdownHeaderTextSplitter(headers_to_split_on=headers_to_split_on)

        # 执行切分：把一个大字符串切成一个 Document 列表

        # 每个 Document 的 page_content 是标题下的正文，metadata 里有标题信息

        raw_docs = markdown_splitter.split_text(md_content)



        # 【上下文增强：把标题信息注入正文】

        for doc in raw_docs:

            fault_name = doc.metadata.get("fault_name", "")

            if fault_name and fault_name not in doc.page_content:

                doc.page_content = f"【关联故障：{fault_name}】\n{doc.page_content}"

            self.docs.append(doc) # 把处理好的文档加入列表



    # ==========================================

    # Milvus 数据库智能初始化

    # ==========================================

    def _smart_init_milvus(self, force_rebuild: bool):

        """

        【智能初始化数据库】

       

        这个方法解决的核心问题是：每次启动程序时，要不要重新建库？

       

        场景分析：

        - 第一次运行：Milvus 里没有这个集合 → 必须创建 Schema + 插入数据

        - md 文件没改：集合已存在 → 直接复用（跳过建库，秒级启动）

        - md 文件改了：集合已存在但数据过时 → 销毁旧集合 + 重新建库

        """

        # has_collection：检查 Milvus 中是否已经存在名为 collection_name 的集合

        # 类比 MySQL 的 "SHOW TABLES LIKE 'fault_inject'"

        if self.client.has_collection(collection_name=self.collection_name):

            if force_rebuild:

                # 用户指定了强制重建（说明 md 文件有更新）

                print(f"[RAG] 收到强制重建指令，正在摧毁旧集合 '{self.collection_name}'...")

                # drop_collection：彻底删除集合及其所有数据和索引

                # 类比 MySQL 的 "DROP TABLE fault_inject"

                self.client.drop_collection(collection_name=self.collection_name)

            else:

                # 不需要重建，直接加载已有集合到内存中

                print(f"[RAG] 检测到集合 '{self.collection_name}' 已存在且未要求重建，直接复用！")

                # load_collection：把集合数据从磁盘加载到 Milvus 的内存中

                # Milvus 的搜索是在内存中进行的（所以快），但数据持久化在磁盘上

                self.client.load_collection(collection_name=self.collection_name)

                return  # 直接返回，跳过后面的建表和插数据逻辑



        # 【核心概念：Schema（表结构定义）】

        # Schema 定义了这个集合（表）里有哪些字段、每个字段是什么类型

        # 类比 MySQL 的 CREATE TABLE 语句

        # Milvus 的集合和 MySQL 的表的对比：

        #   MySQL 表 ↔ Milvus 集合（Collection）

        #   MySQL 行 ↔ Milvus 实体（Entity）

        #   MySQL 列 ↔ Milvus 字段（Field）

        #   MySQL 索引 ↔ Milvus 向量索引（Vector Index）

        print(f"[RAG] 正在创建全新 Schema 并准备写入数据...")

       

        # create_schema 创建一个空的表结构定义对象

        # auto_id=True：主键由 Milvus 自动生成（类比 MySQL 的 AUTO_INCREMENT）

        # enable_dynamic_field=False：不允许插入 Schema 之外的字段（严格模式）

        schema = MilvusClient.create_schema(auto_id=True, enable_dynamic_field=False)

       

        # 【字段 1：id —— 主键】

        # 类比 MySQL 的 "id BIGINT PRIMARY KEY AUTO_INCREMENT"

        # DataType.INT64：64位整数，可以存到 9.2 × 10^18（足够用了）

        # is_primary=True：标记为主键（每个集合必须有且只有一个主键）

        schema.add_field(field_name="id", datatype=DataType.INT64, is_primary=True)

       

        # 【字段 2：vector —— 向量字段（核心！）】

        # 这就是存"语义指纹"的字段，每条数据对应一个 1024 维的浮点数数组

        # DataType.FLOAT_VECTOR：浮点向量类型

        # dim=1024：向量维度，必须和 Embedding 模型的输出维度一致（BGE-M3 输出 1024 维）

        # 如果 dim 写错了（比如写成 768），插入数据时会报维度不匹配的错误

        schema.add_field(field_name="vector", datatype=DataType.FLOAT_VECTOR, dim=self.embedding_dim)

       

        # 【字段 3：text —— 原文字段】

        # 存储文档的原始文本内容（切分后的每个小块）

        # DataType.VARCHAR：变长字符串，类比 MySQL 的 VARCHAR

        # max_length=65535：最大长度 65535 字节（约 2 万汉字，足够一个文档块了）

        # 为什么要存原文？因为向量检索只能找到"最相似的向量"，但最终要返回给用户的是原文！

        schema.add_field(field_name="text", datatype=DataType.VARCHAR, max_length=65535)

       

        # 【字段 4：fault_name —— 元数据字段】

        # 存储故障名称（从 Markdown 标题中提取的元数据）

        # 这个字段不参与向量检索，但在返回结果时可以告诉用户"这段文本属于哪个故障"

        # max_length=1024：故障名称一般不会太长

        schema.add_field(field_name="fault_name", datatype=DataType.VARCHAR, max_length=1024)

       

        # 【向量索引（Index）】

        # 索引是加速搜索的关键。没有索引的话，每次查询都要遍历所有向量算距离（暴力搜索）

        index_params = self.client.prepare_index_params() # 创建一个空的索引配置对象

        index_params.add_index(field_name="vector", metric_type="COSINE", index_type="AUTOINDEX")

       

        # 【创建集合】—— 相当于执行 CREATE TABLE + CREATE INDEX

        # 把 Schema（表结构）和 index_params（索引配置）一起传给 Milvus

        # 执行后，Milvus 中就有了一个空的集合，等待数据写入

        self.client.create_collection(

            collection_name=self.collection_name,  # 集合名称

            schema=schema,                          # 表结构定义

            index_params=index_params               # 索引配置

        )

        # 表建好了，接下来往里面塞数据

        self._insert_docs_to_milvus()



    # ==========================================

    # 数据写入 Milvus

    # ==========================================

    def _insert_docs_to_milvus(self):

        """

        【把文档向量化并写入 Milvus 数据库】

       

        这一步做的事情（类比 MySQL）：

        1. 对每个文档调用 BGE-M3 模型生成 1024 维向量  → 准备 INSERT 的数据

        2. 把 {向量, 原文, 故障名称} 打包成字典列表     → 拼 INSERT INTO ... VALUES (...)

        3. 调用 client.insert 写入 Milvus               → 执行 INSERT 语句

        4. 调用 client.load_collection 加载到内存        → 让数据可被检索

        """

        print(f"[RAG] 正在生成向量并同步文档到 Milvus...")

       

        # 从所有文档对象中提取纯文本内容，组成一个字符串列表

        texts = [doc.page_content for doc in self.docs]

       

        # 【耗时操作！】把所有文本批量送入 BGE-M3 模型，生成对应的向量

        # embed_documents 方法会：

        #   1. 对每个文本做 tokenization（分词）

        #   2. 送入 BERT 类模型做前向传播

        #   3. 取 [CLS] token 的输出作为整个文本的向量表征

        #   4. 做 L2 归一化（因为我们设了 normalize_embeddings=True）

        # 返回值 vectors 是一个二维列表：[[0.12, -0.45, ...], [0.33, 0.67, ...], ...]

        # 每个子列表长度为 1024（向量维度）

        vectors = self.embeddings.embed_documents(texts)

       

        # 把每个文档的 {向量, 原文, 元数据} 打包成 Milvus 要求的字典格式

        data = []

        for doc, vector in zip(self.docs, vectors):

            # 每条数据是一个字典，key 对应 Schema 中定义的字段名

            # 注意：没有 "id" 字段，因为我们设了 auto_id=True，Milvus 会自动生成

            data.append({

                "vector": vector,                                    # 1024 维浮点数组

                "text": doc.page_content,                            # 原始文本

                "fault_name": doc.metadata.get("fault_name", "未知")  # 故障名称元数据

            })

           

        # 【执行批量插入】

        # 类比 MySQL 的 INSERT INTO fault_injection_manual (vector, text, fault_name) VALUES (...)

        self.client.insert(collection_name=self.collection_name, data=data)

       

        # 【加载集合到内存】

        # load 之后，Milvus 会在内存中构建索引结构（如 HNSW 图），使得搜索可以在毫秒级完成

        self.client.load_collection(collection_name=self.collection_name)

        print("[RAG] 同步完成！")



    # ==========================================

    # 双路召回 + 重排 的核心搜索方法

    # ==========================================

    def search(self, query: str, ensemble_k: int = 10, final_k: int = 2) -> str:

        """

        【混合检索 + 重排：完整的搜索流水线】

       

        这是 RAG 系统最核心的方法，实现了工业界主流的"双路召回 + 交叉重排"架构。

       

        为什么要双路召回？

        - BM25（关键词匹配）擅长：精确关键词搜索，比如搜"OSPF"就能精准找到含 OSPF 的文档

        - 向量检索（语义匹配）擅长：理解同义词，比如搜"路由协议故障"也能找到 OSPF 相关文档

        - 单独用任一路都有盲区，两路合起来互补，大大降低"漏掉正确答案"的概率

       

        为什么要重排？

        - 粗排阶段（BM25 + 向量检索）追求"召回率"（别漏掉），但排序不一定准

        - 重排阶段（CrossEncoder）追求"精准率"（挑最好的），用更重的模型精细打分

        - 这是"漏斗型"架构：全部文档 → 粗排取 20 个 → 重排取 1-2 个 → 喂给大模型

       

        参数说明：

        - query: 用户的问题，比如 "请你把将接口配置为被动"

        - ensemble_k: 每路粗排各取多少个候选（默认 10）

            设太小（如 3）→ 可能漏掉正确答案

            设太大（如 50）→ 后续重排要对 50 个文档打分，会很慢

            建议 10-20

        - final_k: 重排后最终取多少个结果返回给大模型（默认 1）

            设为 1 → 只给大模型最相关的 1 段文档，减少干扰，适合精确问答

            设为 2-3 → 给大模型多几段参考，适合需要综合多段信息的场景

       

        返回值：格式化的文本字符串，直接拼到大模型的 prompt 中作为上下文

        """

       

        # ==========================================

        # 【召回路 1】BM25 稀疏检索（关键词精准匹配）

        # ==========================================

        # BM25 的工作原理：

        #   1. 对 query 分词：比如"接口配置被动" → ["接口", "配置", "被动"]

        #   2. 对每个文档，统计这些词出现的频率（TF，Term Frequency）

        #   3. 考虑词的稀有程度（IDF，Inverse Document Frequency）

        #      "的" 出现在 90% 的文档中 → IDF 低 → 权重小

        #      "OSPF" 只出现在 2% 的文档中 → IDF 高 → 权重大

        #   4. 综合 TF 和 IDF 计算每个文档的 BM25 得分，取得分最高的 k 个

        # 注意：BM25Retriever 是纯内存操作，不需要数据库，每次搜索时临时创建也可以

        bm25_retriever = BM25Retriever.from_documents(self.docs)

        bm25_retriever.k = ensemble_k  # 设置返回的文档数量（取 BM25 得分最高的 10 个）

        bm25_docs = bm25_retriever.invoke(query)

       

        # ==========================================

        # 【召回路 2】Milvus 稠密向量检索（语义模糊匹配）

        # ==========================================

        # 稠密检索的工作原理：

        #   1. 把用户的 query 也用 BGE-M3 模型转成 1024 维向量

        #   2. 在 Milvus 中找到与 query 向量余弦相似度最高的 k 个文档向量

        #   3. 返回这些向量对应的原文和元数据

        query_vector = self.embeddings.embed_query(query)

       

        # 【执行向量搜索】

        # 类比 MySQL 的 SELECT text, fault_name FROM ... ORDER BY cosine_similarity(vector, query_vector) DESC LIMIT 10

        milvus_results = self.client.search(

            collection_name=self.collection_name,  # 在哪个集合中搜索

            data=[query_vector],      # 查询向量（注意是一个列表，支持批量查询，所以要套一层 []）

            limit=ensemble_k,         # 返回最相似的 10 个结果

            output_fields=["text", "fault_name"]  # 搜索结果中需要返回哪些字段

            # 如果不指定 output_fields，默认只返回 id 和距离分数，拿不到原文

        )

       

        # 【解析 Milvus 搜索结果】

        # milvus_results 的数据结构是一个二维列表：

        # [[hit1, hit2, hit3, ...]]  ← 外层列表对应每个查询（我们只有 1 个查询，所以外层只有 1 个元素）

        # 每个 hit 是一个字典，结构类似：

        # {"id": 42, "distance": 0.8765, "entity": {"text": "...", "fault_name": "CPU过载"}}

        milvus_docs = []

        for hits in milvus_results:         # 遍历每个查询的结果（这里只有 1 个查询）

            for hit in hits:                # 遍历该查询的每个命中结果

                entity = hit.get("entity")  # 取出实体数据（包含我们要的 text 和 fault_name）

                # 把 Milvus 的结果转换成 LangChain 的 Document 对象，统一数据格式

                milvus_docs.append(Document(

                    page_content=entity.get("text"),                      # 文档原文

                    metadata={"fault_name": entity.get("fault_name")}     # 元数据

                ))

       

        # ==========================================

        # 【合并去重】

        # ==========================================

        # 去重策略：用文本内容（page_content）作为字典的 key

        # Python 字典天然去重：如果两个 Document 的 page_content 完全一样，后者会覆盖前者

        # bm25_docs + milvus_docs：先拼成一个大列表

        unique_docs = {doc.page_content: doc for doc in (bm25_docs + milvus_docs)}

        # .values()：取去重后的所有 Document 对象

        candidate_docs = list(unique_docs.values())

       

        # 如果两路都没找到任何文档

        if not candidate_docs:

            return "检索系统未找到与该故障相关的信息。"



        # ==========================================

        # 【交叉重排（Reranking）—— 精中选精】

        # ==========================================

        # CrossEncoder 的工作原理：

        #   输入：[query, doc_text] 这样一对文本

        #   内部：把 query 和 doc_text 用 [SEP] 拼接，送入 BERT 模型

        #         模型能同时"看到"问题和文档，做深层的词与词之间的注意力交互

        #   输出：一个浮点数分数，表示这对 (query, doc) 的相关程度

        #         分数越高 → 越相关

       

        # 构造 [query, doc_text] 配对列表

        pairs = [[query, doc.page_content] for doc in candidate_docs]

       

        # 批量预测相关性分数

        scores = self.cross_encoder.predict(pairs)

       

        # 把文档和对应的分数打包在一起，然后按分数从高到低排序

        # zip(candidate_docs, scores) → [(doc1, 0.92), (doc2, 0.15), (doc3, 0.78)]

        # sorted(..., key=lambda x: x[1], reverse=True) → [(doc1, 0.92), (doc3, 0.78), (doc2, 0.15)]

        scored_docs = sorted(zip(candidate_docs, scores), key=lambda x: x[1], reverse=True)

       

        # ==========================================

        # 【取 Top-K 结果，格式化输出】

        # ==========================================

        # scored_docs[:final_k] → 取排序后的前 final_k 个（默认取 1 个，即最最相关的那段文档）

        # 把结果格式化成可读的字符串，后续会拼到大模型的 prompt 中作为上下文参考

        final_result = ""

        for i, (doc, score) in enumerate(scored_docs[:final_k]):

            # 打印调试信息：看看重排后每个文档的置信度得分

            # 一般来说，得分 > 0.5 的文档相关性较高，> 0.8 的高度相关

           

            # print(f"[DEBUG] Top {i+1} 重排置信度得分: {score:.4f}")



            # 格式化输出：包含故障名称和文档内容

            final_result += f"### 匹配结果 [{i+1}] 重排置信度得分: {score:.4f} (故障名称: {doc.metadata.get('fault_name', '未知')})\n"

            final_result += f"{doc.page_content}\n\n"

           

        # 去掉末尾多余的空白，返回最终结果

        return final_result.strip()



    def _drop_all_collections(self):

        """

        丢弃 Milvus 数据库中的全部 collection。

       

        工作原理:

        1. 调用 client.list_collections() 获取当前数据库中所有的集合名称列表。

        2. 遍历该列表，逐个调用 client.drop_collection() 进行删除。

        """

        # 获取当前数据库下所有的集合名称列表 (例如: ['fault_injection_manual', 'another_collection'])

        collection_names = self.client.list_collections()

       

        if not collection_names:

            print("[Milvus] 当前数据库为空，没有需要删除的 collection。")

            return



        # 遍历并逐一删除

        for name in collection_names:

            self.client.drop_collection(collection_name=name)

            print(f"[Milvus] 已成功删除 collection: {name}")
